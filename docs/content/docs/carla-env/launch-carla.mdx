### Settings

<GridContainer>
    <Image
        alt="settings-window"
        src="/carla-env/launch-carla/settings.png"
        width={400}
        height={400}
        className="dark:invert border border-muted-foreground rounded-sm"
    />

<ParameterAccordion
    title="Carla Path"
    subtitle="Carla Simulator executable file"
    types="string"
    description="Specify the full path to the Carla Simulator executable file. This is necessary to launch the Carla Simulator correctly."
/>

<ParameterAccordion
    title="Simulation Mode"
    subtitle="Choose the simulation mode"
    types="Synchronous (default) | Asynchronous"
    description={
        <div>
            Select the mode of simulation. In{" "}
            <Link
                target="_blank"
                href="https://carla.readthedocs.io/en/latest/adv_traffic_manager/#synchronous-mode"
                className="font-bold"
            >
                Synchronous
            </Link>{" "}
            mode, the simulation is stepped according to the client commands,
            ensuring precise control over the simulation time steps. In{" "}
            <span className="font-bold">Asynchronous</span> mode, the simulation
            runs freely without waiting for the client commands, which is useful
            for real-time simulations but less precise in control.

            <Callout title="Warning" type="warn">
                Add{" "}
                <Link href="/docs/carla-env/world-tick" className="font-bold">World Tick</Link> {" "}
                block when using sync mode.
            </Callout>
        </div>
    }

/>

<ParameterAccordion
    title="Quality"
    subtitle="Set the graphics quality"
    types="Epic (default) | Low"
    description="Select the desired graphics quality for the Carla Simulator. Higher quality settings provide better visuals but require more computational resources."
/>
